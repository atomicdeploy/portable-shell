cmake_minimum_required(VERSION 3.16)
project(PortableShell VERSION 0.13.0 LANGUAGES C)
option(PORTABLE_SHELL_USE_LIBC "Use the host C library for memory/string primitives" OFF)
option(PORTABLE_SHELL_PROVIDE_MEMORY "Supply compiler-required memory symbols in portable mode" ON)
option(PORTABLE_SHELL_AUDIT_SYMBOLS "Audit archive external dependencies after building" ON)
set(PORTABLE_SHELL_OPTIMIZATION "size" CACHE STRING "size, speed, or toolchain")
set_property(CACHE PORTABLE_SHELL_OPTIMIZATION PROPERTY STRINGS size speed toolchain)
option(PORTABLE_SHELL_ASCII_ONLY "Compile with Unicode and color disabled" OFF)
option(PORTABLE_SHELL_BUILD_HOST "Build the optional native terminal/TCP/serial adapter" ON)
option(PORTABLE_SHELL_BUILD_TESTS "Build brief byte-level checks" ON)
option(PORTABLE_SHELL_COMPACT "Explicit smaller buffers; features unchanged" OFF)
option(PORTABLE_SHELL_COMPILER_RT "Include freestanding compiler helpers for Xtensa" OFF)
option(PORTABLE_SHELL_BUILD_LEGACY "Build hosted adapter for the original atomicdeploy API" OFF)
add_library(portable_shell STATIC src/portable_shell.c src/portable_shell_device.c)
if(PORTABLE_SHELL_USE_LIBC)
  target_compile_definitions(portable_shell PRIVATE PS_USE_LIBC)
elseif(PORTABLE_SHELL_PROVIDE_MEMORY)
  target_sources(portable_shell PRIVATE ports/memory_runtime.c)
endif()
include(GNUInstallDirs)
install(TARGETS portable_shell ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/portable_shell.h include/portable_shell_device.h
  include/portable_terminal.h include/portable_editor.h
  include/portable_parse.h include/portable_probe.h
  include/portable_shell_editor.hpp include/portable_device_view.hpp
  include/portable_device_table.hpp include/portable_style.h
  DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(PORTABLE_SHELL_BUILD_LEGACY)
  add_library(portable_shell_legacy STATIC compat/legacy.c host/legacy_io.c)
  install(TARGETS portable_shell_legacy ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  install(FILES include/portable_shell_legacy.h include/ps_io.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  target_link_libraries(portable_shell_legacy PUBLIC portable_shell)
  target_compile_definitions(portable_shell_legacy PUBLIC PS_ENABLE_LEGACY_API=1)
  if(WIN32)
    target_link_libraries(portable_shell_legacy PUBLIC ws2_32)
  endif()
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(portable_shell_legacy PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(PORTABLE_SHELL_BUILD_TESTS AND NOT CMAKE_CROSSCOMPILING)
    enable_testing()
    add_executable(legacy_original_checks tests/test_shell.c)
    target_link_libraries(legacy_original_checks PRIVATE portable_shell_legacy)
    add_test(NAME legacy_original_checks COMMAND legacy_original_checks)
    add_executable(legacy_adapter_checks tests/legacy_checks.c)
    target_link_libraries(legacy_adapter_checks PRIVATE portable_shell_legacy)
    add_test(NAME legacy_adapter_checks COMMAND legacy_adapter_checks)
    add_executable(legacy_transport_checks tests/legacy_io_checks.c)
    target_link_libraries(legacy_transport_checks PRIVATE portable_shell_legacy)
    add_test(NAME legacy_transport_checks COMMAND legacy_transport_checks)
    set_tests_properties(legacy_transport_checks PROPERTIES TIMEOUT 15)
    foreach(check legacy_original_checks legacy_adapter_checks legacy_transport_checks)
      if(MSVC)
        target_compile_options(${check} PRIVATE /UNDEBUG)
      else()
        target_compile_options(${check} PRIVATE -UNDEBUG)
      endif()
    endforeach()
  endif()
  if(UNIX AND NOT CMAKE_CROSSCOMPILING)
    add_executable(legacy_linux_example examples/linux_example.c)
    target_link_libraries(legacy_linux_example PRIVATE portable_shell_legacy)
  endif()
endif()
if(PORTABLE_SHELL_COMPILER_RT)
  target_sources(portable_shell PRIVATE ports/compiler_runtime.c)
endif()
if(PORTABLE_SHELL_COMPACT)
  target_compile_definitions(portable_shell PUBLIC PS_LINE=128 PS_HISTORY=4 PS_ROWS=4 PS_COMMANDS=24)
endif()
target_include_directories(portable_shell PUBLIC include)
target_compile_features(portable_shell PUBLIC c_std_99)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(portable_shell PRIVATE -Wall -Wextra -Wpedantic -Werror -fstack-usage)
  if(PORTABLE_SHELL_OPTIMIZATION STREQUAL "size")
    if(CMAKE_C_COMPILER_ID MATCHES "Clang")
      target_compile_options(portable_shell PRIVATE -Oz)
    else()
      target_compile_options(portable_shell PRIVATE -Os)
    endif()
  elseif(PORTABLE_SHELL_OPTIMIZATION STREQUAL "speed")
    target_compile_options(portable_shell PRIVATE -O3)
  elseif(NOT PORTABLE_SHELL_OPTIMIZATION STREQUAL "toolchain")
    message(FATAL_ERROR "PORTABLE_SHELL_OPTIMIZATION must be size, speed, or toolchain")
  endif()
  if(NOT PORTABLE_SHELL_USE_LIBC)
    target_compile_options(portable_shell PRIVATE -ffreestanding -fno-builtin -fno-stack-protector)
  endif()
  set_source_files_properties(ports/memory_runtime.c PROPERTIES
    COMPILE_OPTIONS "-ffreestanding;-fno-builtin;-fno-lto;-fno-stack-protector")
  if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    set_property(SOURCE ports/memory_runtime.c APPEND PROPERTY COMPILE_OPTIONS
      -fno-tree-loop-distribute-patterns)
  endif()
  if(WIN32 AND CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|x86_64" AND NOT PORTABLE_SHELL_USE_LIBC)
    target_compile_options(portable_shell PRIVATE -mno-stack-arg-probe)
  endif()
endif()
if(PORTABLE_SHELL_BUILD_HOST AND NOT CMAKE_CROSSCOMPILING)
  if(WIN32)
    add_executable(portable_shell_host host/windows.c)
    if(PORTABLE_SHELL_USE_LIBC)
      target_compile_definitions(portable_shell_host PRIVATE PS_HOSTED_CRT)
      if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
        target_link_options(portable_shell_host PRIVATE -s)
      endif()
    elseif(CMAKE_C_COMPILER_ID STREQUAL "GNU")
      target_compile_options(portable_shell_host PRIVATE -Os -ffreestanding -fno-builtin
        -fno-stack-protector -mno-stack-arg-probe)
      target_link_options(portable_shell_host PRIVATE -nostdlib -Wl,-e,host_entry -Wl,--subsystem,console -s)
    endif()
    target_link_libraries(portable_shell_host PRIVATE portable_shell kernel32 ws2_32)
  elseif(UNIX)
    add_executable(portable_shell_host host/posix.c)
    target_compile_options(portable_shell_host PRIVATE -Os -Wall -Wextra -Wpedantic)
    target_link_libraries(portable_shell_host PRIVATE portable_shell)
  endif()
  if(TARGET portable_shell_host)
    set_target_properties(portable_shell_host PROPERTIES OUTPUT_NAME PortableShell)
  endif()
endif()
if(PORTABLE_SHELL_ASCII_ONLY)
  target_compile_definitions(portable_shell PRIVATE PS_ASCII_ONLY)
  if(TARGET portable_shell_legacy)
    target_compile_definitions(portable_shell_legacy PRIVATE PS_ASCII_ONLY)
  endif()
endif()
if(PORTABLE_SHELL_BUILD_TESTS AND NOT CMAKE_CROSSCOMPILING)
  enable_testing()
  enable_language(CXX)
  add_executable(terminal_replay tests/ansi_render.c)
  target_link_libraries(terminal_replay PRIVATE portable_shell)
  add_executable(micro_checks tests/micro_checks.c)
  target_include_directories(micro_checks PRIVATE include)
  if(MSVC)
    target_compile_options(micro_checks PRIVATE /UNDEBUG)
  else()
    target_compile_options(micro_checks PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME micro_checks COMMAND micro_checks)
  foreach(check event_editor_checks device_view_checks device_table_checks)
    add_executable(${check} tests/${check}.cpp)
    target_include_directories(${check} PRIVATE include)
    target_compile_features(${check} PRIVATE cxx_std_11)
    set_target_properties(${check} PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    if(PORTABLE_SHELL_USE_LIBC)
      target_compile_definitions(${check} PRIVATE PS_TERMINAL_USE_LIBC)
    endif()
    if(MSVC)
      target_compile_options(${check} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${check} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    endif()
    add_test(NAME ${check} COMMAND ${check})
  endforeach()
  if(PORTABLE_SHELL_USE_LIBC)
    add_executable(portable_shell_checks tests/checks.c tests/hosted_entry.c)
    target_compile_definitions(portable_shell_checks PRIVATE PS_USE_LIBC)
  elseif(WIN32 AND CMAKE_C_COMPILER_ID STREQUAL "GNU")
    add_executable(portable_shell_checks tests/checks.c tests/entry.c)
    target_compile_options(portable_shell_checks PRIVATE -Os -ffreestanding -fno-builtin
      -fno-stack-protector -mno-stack-arg-probe)
    target_link_options(portable_shell_checks PRIVATE -nostdlib -Wl,-e,test_entry)
  elseif(UNIX)
    add_executable(portable_shell_checks tests/checks.c tests/posix_entry.c)
  endif()
  if(TARGET portable_shell_checks)
    target_sources(portable_shell_checks PRIVATE tests/runtime_checks.c tests/memory_checks.c tests/terminal_checks.c)
    if(NOT PORTABLE_SHELL_COMPILER_RT)
      target_sources(portable_shell_checks PRIVATE ports/compiler_runtime.c)
    endif()
    target_link_libraries(portable_shell_checks PRIVATE portable_shell)
    add_test(NAME byte_checks COMMAND portable_shell_checks)
  endif()
endif()
if(PORTABLE_SHELL_AUDIT_SYMBOLS)
  if(NOT CMAKE_NM)
    message(FATAL_ERROR "Symbol audit needs CMAKE_NM; set it to a compatible nm or disable audit explicitly")
  endif()
  add_custom_target(portable_shell_symbol_audit ALL
    COMMAND ${CMAKE_COMMAND} "-DNM=${CMAKE_NM}" "-DARCHIVE=$<TARGET_FILE:portable_shell>"
      "-DUSE_LIBC=${PORTABLE_SHELL_USE_LIBC}" "-DPROVIDE_MEMORY=${PORTABLE_SHELL_PROVIDE_MEMORY}"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/check-symbols.cmake"
    DEPENDS portable_shell VERBATIM)
endif()
