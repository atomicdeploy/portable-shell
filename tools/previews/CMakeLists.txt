cmake_minimum_required(VERSION 3.20)
project(PortableShellPreviews LANGUAGES C CXX)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(workspace_preview "${root}/tests/export_previews.c" "${root}/tests/render.c"
  "${root}/src/portable_shell.c" "${root}/src/portable_shell_device.c")
add_executable(full_preview "${root}/tests/render_device_profiles.cpp" "${root}/tests/render.c")
add_executable(reduced_preview "${root}/tests/render_device_profiles.cpp" "${root}/tests/render.c")
target_compile_definitions(full_preview PRIVATE PS_REPLAY_ONLY)
target_compile_definitions(reduced_preview PRIVATE PS_REPLAY_ONLY PS_DEVICE_VIEW_TABLE=0 PS_DEVICE_VIEW_WIDGETS=0
  PS_DEVICE_VIEW_DIALOGS=0 PS_DEVICE_VIEW_MATRIX=0 PS_STYLE_THEMES=0 PS_DEVICE_VIEW_VERTICAL_TABS=0)
foreach(target workspace_preview full_preview reduced_preview)
  target_include_directories(${target} PRIVATE "${root}/include")
  target_compile_features(${target} PRIVATE c_std_99 cxx_std_11)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  endforeach()
endforeach()
